<script type="module">
  import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
  import { 
    getAuth, 
    signInWithEmailAndPassword, 
    createUserWithEmailAndPassword, 
    updateProfile, 
    onAuthStateChanged, 
    signOut,
    signInWithCustomToken
  } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
  import { 
    getFirestore, 
    collection, 
    doc, 
    setDoc, 
    addDoc, 
    onSnapshot, 
    serverTimestamp
  } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

  // Custom Firebase configuration for nexus-chat-b08ad
  const firebaseConfig = {
    apiKey: "AIzaSyAA3DH6J7rAbJpJ0S8HWpcbJOhIarC9aa8",
    authDomain: "nexus-chat-b08ad.firebaseapp.com",
    projectId: "nexus-chat-b08ad",
    storageBucket: "nexus-chat-b08ad.firebasestorage.app",
    messagingSenderId: "288153271485",
    appId: "1:288153271485:web:feb594a566e1b8ccd04bd4",
    measurementId: "G-JLERQS6WWE"
  };

  const app = initializeApp(firebaseConfig);
  const auth = getAuth(app);
  const db = getFirestore(app);

  // Application State
  let currentUser = null;
  let currentChannel = 'general';
  let isSignupMode = false;
  let unsubscribeMessages = null;
  let unsubscribeMembers = null;

  const CHANNELS = [
    { id: 'general', name: 'general', desc: 'General developer discussion and chat.' },
    { id: 'lounge', name: 'lounge', desc: 'Off-topic discussions and relaxed talk.' },
    { id: 'gaming', name: 'gaming', desc: 'Game recommendations, setups, and streaming.' },
    { id: 'announcements', name: 'announcements', desc: 'Official updates and channel news.' }
  ];

  // Firestore Document / Collection Helpers
  const getPublicDocPath = (colName, docId) => doc(db, colName, docId);
  const getPublicColPath = (colName) => collection(db, colName);

  // 1. Initial Authentication Handler
  async function initAuth() {
    try {
      if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
        await signInWithCustomToken(auth, __initial_auth_token);
      }
    } catch (e) {
      console.warn("Custom token login failed, falling back to standard auth state listener", e);
    }

    onAuthStateChanged(auth, async (user) => {
      if (user) {
        currentUser = user;
        document.getElementById('auth-screen').classList.add('hidden');
        updateUserProfileUI();
        await registerPresence(user);
        renderChannels();
        switchChannel(currentChannel);
        listenToActiveMembers();
      } else {
        currentUser = null;
        document.getElementById('auth-screen').classList.remove('hidden');
      }
    });
  }

  // 2. Auth Form Handler (Login / Signup)
  window.handleAuthSubmit = async function(e) {
    e.preventDefault();
    clearAuthError();

    const email = document.getElementById('auth-email').value;
    const password = document.getElementById('auth-password').value;
    const username = document.getElementById('auth-username').value;

    try {
      if (isSignupMode) {
        if (!username.trim()) {
          showAuthError("Please enter a username.");
          return;
        }
        const userCred = await createUserWithEmailAndPassword(auth, email, password);
        await updateProfile(userCred.user, { displayName: username });
      } else {
        await signInWithEmailAndPassword(auth, email, password);
      }
    } catch (err) {
      showAuthError(err.message.replace("Firebase: ", ""));
    }
  };

  // Toggle Login/Signup UI
  window.toggleAuthMode = function() {
    isSignupMode = !isSignupMode;
    const title = document.getElementById('auth-title');
    const subtitle = document.getElementById('auth-subtitle');
    const usernameGroup = document.getElementById('username-field-group');
    const submitBtn = document.getElementById('auth-submit-btn');
    const togglePrompt = document.getElementById('auth-toggle-prompt');
    const toggleBtn = document.getElementById('auth-toggle-btn');

    if (isSignupMode) {
      title.textContent = "Create an account";
      subtitle.textContent = "Join our real-time developer community today!";
      usernameGroup.classList.remove('hidden');
      submitBtn.textContent = "Continue";
      togglePrompt.textContent = "Already have an account?";
      toggleBtn.textContent = "Log In";
    } else {
      title.textContent = "Welcome Back!";
      subtitle.textContent = "We're so excited to see you again!";
      usernameGroup.classList.add('hidden');
      submitBtn.textContent = "Log In";
      togglePrompt.textContent = "Need an account?";
      toggleBtn.textContent = "Register";
    }
  };

  function showAuthError(msg) {
    const errorBox = document.getElementById('auth-error');
    document.getElementById('auth-error-msg').textContent = msg;
    errorBox.classList.remove('hidden');
  }

  function clearAuthError() {
    document.getElementById('auth-error').classList.add('hidden');
  }

  window.handleLogout = function() {
    signOut(auth);
  };

  // 3. User Presence & Profiles
  async function registerPresence(user) {
    if (!user) return;
    try {
      const userDocRef = getPublicDocPath('presence', user.uid);
      await setDoc(userDocRef, {
        uid: user.uid,
        displayName: user.displayName || user.email.split('@')[0],
        email: user.email,
        lastSeen: serverTimestamp(),
        online: true
      }, { merge: true });
    } catch (e) {
      console.error("Presence registration error:", e);
    }
  }

  function updateUserProfileUI() {
    if (!currentUser) return;
    const name = currentUser.displayName || currentUser.email.split('@')[0];
    document.getElementById('user-display-name').textContent = name;
    document.getElementById('user-avatar').textContent = name.charAt(0).toUpperCase();
  }

  // 4. Channel Rendering & Navigation
  function renderChannels() {
    const channelListEl = document.getElementById('channel-list');
    channelListEl.innerHTML = `
      <div class="text-xs font-bold text-discordTextMuted px-2 mb-1 uppercase tracking-wider flex items-center justify-between">
        <span>Text Channels</span>
      </div>
    `;

    CHANNELS.forEach(ch => {
      const isActive = ch.id === currentChannel;
      const btn = document.createElement('div');
      btn.className = `flex items-center gap-2 px-2 py-1.5 rounded cursor-pointer transition ${
        isActive ? 'bg-discordHover text-white font-semibold' : 'text-discordTextMuted hover:bg-discordHover hover:text-gray-200'
      }`;
      btn.innerHTML = `
        <i class="fa-solid fa-hashtag text-gray-500 text-sm"></i>
        <span class="text-sm truncate">${ch.name}</span>
      `;
      btn.onclick = () => switchChannel(ch.id);
      channelListEl.appendChild(btn);
    });
  }

  function switchChannel(channelId) {
    currentChannel = channelId;
    const selected = CHANNELS.find(c => c.id === channelId);
    
    document.getElementById('active-channel-name').textContent = selected.name;
    document.getElementById('channel-desc').textContent = selected.desc;
    document.getElementById('message-input').placeholder = `Message #${selected.name}`;

    renderChannels();
    listenToMessages(channelId);
  }

  // 5. Real-Time Chat Subscriptions
  function listenToMessages(channelId) {
    if (unsubscribeMessages) unsubscribeMessages();

    const messagesContainer = document.getElementById('messages-container');
    messagesContainer.innerHTML = '<div class="text-center text-xs text-discordTextMuted py-4">Loading messages...</div>';

    const msgCollection = getPublicColPath('messages');

    unsubscribeMessages = onSnapshot(msgCollection, (snapshot) => {
      const messages = [];
      snapshot.forEach(doc => {
        const data = doc.data();
        if (data.channelId === channelId) {
          messages.push({ id: doc.id, ...data });
        }
      });

      // Sort by timestamp ascending
      messages.sort((a, b) => {
        const t1 = a.timestamp?.toMillis ? a.timestamp.toMillis() : (a.timestamp || 0);
        const t2 = b.timestamp?.toMillis ? b.timestamp.toMillis() : (b.timestamp || 0);
        return t1 - t2;
      });

      renderMessages(messages);
    }, (error) => {
      console.error("Messages fetch failed:", error);
    });
  }

  function renderMessages(messages) {
    const container = document.getElementById('messages-container');
    if (messages.length === 0) {
      container.innerHTML = `
        <div class="flex flex-col items-center justify-center h-full text-discordTextMuted">
          <i class="fa-solid fa-comments text-4xl mb-2"></i>
          <p class="font-semibold text-white">Welcome to #${currentChannel}!</p>
          <p class="text-xs">This is the start of the #${currentChannel} channel.</p>
        </div>
      `;
      return;
    }

    container.innerHTML = '';
    messages.forEach(msg => {
      const timeStr = msg.timestamp?.seconds 
        ? new Date(msg.timestamp.seconds * 1000).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
        : 'Just now';

      const avatarLetter = (msg.senderName || 'U').charAt(0).toUpperCase();

      const msgEl = document.createElement('div');
      msgEl.className = "flex items-start gap-3 hover:bg-[#2e3035] -mx-4 px-4 py-1.5 transition rounded";
      msgEl.innerHTML = `
        <div class="w-10 h-10 rounded-full bg-blurple flex items-center justify-center font-bold text-white shrink-0 mt-0.5">
          ${avatarLetter}
        </div>
        <div class="flex flex-col overflow-hidden">
          <div class="flex items-center gap-2">
            <span class="font-semibold text-white text-sm hover:underline cursor-pointer">${escapeHtml(msg.senderName || 'User')}</span>
            <span class="text-[10px] text-discordTextMuted">${timeStr}</span>
          </div>
          <p class="text-sm text-gray-300 break-words mt-0.5">${escapeHtml(msg.content)}</p>
        </div>
      `;
      container.appendChild(msgEl);
    });

    // Auto scroll to bottom
    container.scrollTop = container.scrollHeight;
  }

  // Send Message Handler
  window.handleSendMessage = async function(e) {
    e.preventDefault();
    const input = document.getElementById('message-input');
    const content = input.value.trim();

    if (!content || !currentUser) return;

    input.value = '';

    try {
      await addDoc(getPublicColPath('messages'), {
        channelId: currentChannel,
        senderUid: currentUser.uid,
        senderName: currentUser.displayName || currentUser.email.split('@')[0],
        content: content,
        timestamp: serverTimestamp()
      });
    } catch (err) {
      console.error("Error sending message:", err);
    }
  };

  // 6. Member List Subscriptions
  function listenToActiveMembers() {
    if (unsubscribeMembers) unsubscribeMembers();

    unsubscribeMembers = onSnapshot(getPublicColPath('presence'), (snapshot) => {
      const members = [];
      snapshot.forEach(doc => {
        members.push(doc.data());
      });

      const membersList = document.getElementById('members-list');
      const header = document.getElementById('online-count-header');
      header.textContent = `Online — ${members.length}`;

      membersList.innerHTML = '';
      members.forEach(member => {
        const item = document.createElement('div');
        item.className = "flex items-center gap-2 px-2 py-1.5 rounded hover:bg-[#35373c]/60 cursor-pointer";
        item.innerHTML = `
          <div class="relative shrink-0">
            <div class="w-8 h-8 rounded-full bg-blurple flex items-center justify-center text-xs font-bold text-white">
              ${(member.displayName || 'U').charAt(0).toUpperCase()}
            </div>
            <div class="w-2.5 h-2.5 bg-discordOnline border-2 border-discordSidebarBg rounded-full absolute -bottom-0.5 -right-0.5"></div>
          </div>
          <div class="flex flex-col truncate">
            <span class="text-xs font-medium text-gray-200 truncate">${escapeHtml(member.displayName || 'User')}</span>
          </div>
        `;
        membersList.appendChild(item);
      });
    }, (error) => {
      console.error("Presence update failed:", error);
    });
  }

  // 7. Interactive UI Helpers
  window.toggleEmojiPicker = function() {
    const picker = document.getElementById('emoji-picker');
    picker.classList.toggle('hidden');
  };

  window.insertEmoji = function(emoji) {
    const input = document.getElementById('message-input');
    input.value += emoji;
    document.getElementById('emoji-picker').classList.add('hidden');
    input.focus();
  };

  window.toggleMembersList = function() {
    const sidebar = document.getElementById('members-sidebar');
    sidebar.classList.toggle('hidden');
  };

  function escapeHtml(str) {
    return String(str)
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }

  // Initialize Application on Window Load
  window.addEventListener('DOMContentLoaded', () => {
    initAuth();
  });
</script>
